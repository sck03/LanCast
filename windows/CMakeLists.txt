cmake_minimum_required(VERSION 3.25)
project(LanCast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
FetchContent_MakeAvailable(json)
set(GST_ROOT "$ENV{GSTREAMER_1_0_ROOT_MSVC_X86_64}" CACHE PATH "GStreamer MSVC x64 SDK")
if(NOT EXISTS "${GST_ROOT}/include/gstreamer-1.0/gst/gst.h")
    message(FATAL_ERROR "Install the pinned GStreamer runtime and development SDK; set GST_ROOT")
endif()
add_executable(LanCast WIN32 src/main.cpp src/media.cpp)
target_include_directories(LanCast PRIVATE ../core/include "${GST_ROOT}/include/gstreamer-1.0" "${GST_ROOT}/include/glib-2.0" "${GST_ROOT}/lib/glib-2.0/include")
target_link_directories(LanCast PRIVATE "${GST_ROOT}/lib")
target_compile_definitions(LanCast PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN GST_USE_UNSTABLE_API)
target_compile_options(LanCast PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(LanCast PRIVATE nlohmann_json::nlohmann_json
    "${CMAKE_CURRENT_SOURCE_DIR}/../target/x86_64-pc-windows-msvc/release/lancast_core.dll.lib"
    gstreamer-1.0 gstwebrtc-1.0 gstsdp-1.0 gobject-2.0 glib-2.0 ole32 comdlg32 comctl32)
add_custom_command(TARGET LanCast POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/../target/x86_64-pc-windows-msvc/release/lancast_core.dll" "$<TARGET_FILE_DIR:LanCast>")
