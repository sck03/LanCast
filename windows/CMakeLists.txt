cmake_minimum_required(VERSION 3.25)
project(LanCast LANGUAGES C CXX RC)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../build-config.json" LC_DEFAULTS)
string(JSON LC_DEFAULT_VERSION GET "${LC_DEFAULTS}" version)
string(JSON LC_DEFAULT_BUILD GET "${LC_DEFAULTS}" build_number)
if(DEFINED ENV{LC_VERSION} AND NOT "$ENV{LC_VERSION}" STREQUAL "")
    set(LC_DEFAULT_VERSION "$ENV{LC_VERSION}")
endif()
if(DEFINED ENV{LC_BUILD_NUMBER} AND NOT "$ENV{LC_BUILD_NUMBER}" STREQUAL "")
    set(LC_DEFAULT_BUILD "$ENV{LC_BUILD_NUMBER}")
endif()
set(LC_APP_VERSION "${LC_DEFAULT_VERSION}" CACHE STRING "Application major.minor.patch")
set(LC_BUILD_NUMBER "${LC_DEFAULT_BUILD}" CACHE STRING "Application build number")
if(NOT LC_APP_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$" OR NOT LC_BUILD_NUMBER MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "Invalid application version/build number")
endif()
string(REPLACE "." "," LC_VERSION_COMMAS "${LC_APP_VERSION}")
configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
enable_testing()
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
FetchContent_MakeAvailable(json)
add_subdirectory(media)
add_executable(LanCast WIN32 src/main.cpp src/media.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
add_dependencies(LanCast lancast_rtc)
target_include_directories(LanCast PRIVATE ../core/include ../media-native/include)
target_compile_definitions(LanCast PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(LanCast PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(LanCast PRIVATE nlohmann_json::nlohmann_json
    "${CMAKE_CURRENT_SOURCE_DIR}/../target/x86_64-pc-windows-msvc/release/lancast_core.dll.lib"
    ole32 shell32 comdlg32 comctl32)
add_custom_command(TARGET LanCast POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/../target/x86_64-pc-windows-msvc/release/lancast_core.dll" "$<TARGET_FILE_DIR:LanCast>")
