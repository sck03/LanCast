cmake_minimum_required(VERSION 3.22)
project(LanCastMedia LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(FFMPEG_ROOT "" CACHE PATH "Explicit minimal FFmpeg install prefix")
if(NOT EXISTS "${FFMPEG_ROOT}/include/libavformat/avformat.h")
  message(FATAL_ERROR "Build scripts/build-ffmpeg.py first and set FFMPEG_ROOT")
endif()
if(WIN32)
  add_library(lancast_ts SHARED ts/mux.cpp video/h264.cpp ts/exports.def)
  set_target_properties(lancast_ts PROPERTIES PREFIX "")
  # Linked FFmpeg/GCC archives can carry debug data even in a Release build.
  target_link_options(lancast_ts PRIVATE -static-libgcc -static-libstdc++ "$<$<CONFIG:Release,MinSizeRel>:-s>")
else()
  add_library(lancast_ts STATIC ts/mux.cpp video/h264.cpp)
endif()
target_include_directories(lancast_ts PUBLIC include PRIVATE "${FFMPEG_ROOT}/include")
foreach(component avformat avcodec avutil)
  find_library(LC_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  target_link_libraries(lancast_ts PUBLIC "${LC_${component}}")
endforeach()
if(WIN32)
  # GNU archive resolution is left-to-right: system imports follow their users.
  target_link_libraries(lancast_ts PRIVATE bcrypt ws2_32 -Wl,-Bstatic winpthread -Wl,-Bdynamic)
endif()
if(UNIX)
  target_link_libraries(lancast_ts PUBLIC m)
endif()
if(ANDROID)
  add_library(lancast_media SHARED ts/android.cpp)
  target_link_libraries(lancast_media PRIVATE lancast_ts log)
  target_link_options(lancast_media PRIVATE "-Wl,-z,max-page-size=16384")
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(mux_contract tests/mux_contract.cpp)
  target_link_libraries(mux_contract PRIVATE lancast_ts)
  add_test(NAME mux_contract COMMAND mux_contract)
  add_executable(h264_fixture tests/h264_fixture.cpp)
  target_link_libraries(h264_fixture PRIVATE lancast_ts)
endif()
